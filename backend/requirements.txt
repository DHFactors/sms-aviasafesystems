# named databases (Client(database=...)). Explicit pin guards the Docker build.

# init with: TypeError: Client.__init__() got an unexpected keyword argument
# 'proxies'. httpx is pinned <0.28.0 to keep the two compatible.

httpx>=0.27.0,<0.28.0
